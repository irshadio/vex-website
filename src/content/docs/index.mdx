## A taste of Vex

```swift
struct User {
    name: string
    age: int = 18

    fn greet() {
        println("Hi, I'm {name}")
    }

    fn isAdult() -> bool => age >= 18
}

fn main() {
    let users = [
        User(name: "Ann", age: 30),
        User(name: "Sam"),
    ]

    for i, user in users {
        user.greet()
        println("{i}: adult = {user.isAdult()}")
    }
}
```
